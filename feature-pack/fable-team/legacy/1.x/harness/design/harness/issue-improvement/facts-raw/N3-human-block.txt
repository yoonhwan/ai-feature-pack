last_epoch() {  # $1 session  $2 window_activity  $3 midnight_epoch  $4 now  →  "epoch <:|~>"
  local txt ampm h m t
  txt="$(tmux capture-pane -t "=$1:" -p -S -300 2>/dev/null | tr -s '\n ' '  ')"
  # 한국어 로케일 "done 오후 12:56" / 영어 로케일 "done 12:56 PM" 둘 다
  if [[ "$txt" =~ done\ +(오전|오후)\ +([0-9]{1,2}):([0-9]{2}) ]]; then
    ampm="${BASH_REMATCH[1]}"; h=$((10#${BASH_REMATCH[2]})); m=$((10#${BASH_REMATCH[3]}))
  elif [[ "$txt" =~ done\ +([0-9]{1,2}):([0-9]{2})\ +(AM|PM) ]]; then
    h=$((10#${BASH_REMATCH[1]})); m=$((10#${BASH_REMATCH[2]})); ampm="${BASH_REMATCH[3]}"
  else
    echo "${2:-0} ~"; return
  fi
